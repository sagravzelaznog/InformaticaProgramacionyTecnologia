Prácticas 14 y 15 U2: Sistema de Biblioteca con Clases

Este código define las clases `Libro` y `Estudiante` y simula la acción de prestar y devolver libros, una aplicación práctica de POO.

# practica14 u2 parte 1 de 2: Definición de la clase Libro
class Libro:
    def __init__(self, titulo, autor):
        self.titulo = titulo
        self.autor = autor
        self.prestado = False # Estado inicial: no prestado

    def prestar(self):
        if not self.prestado:
            self.prestado = True
            return True
        return False

    def devolver(self):
        if self.prestado:
            self.prestado = False
            return True
        return False

# practica14 u2 parte 2 de 2: Definición de la clase Estudiante
class Estudiante:
    def __init__(self, nombre, matricula):
        self.nombre = nombre
        self.matricula = matricula
        self.libros_prestados = []

    def tomar_prestado(self, libro):
        if libro.prestar():
            self.libros_prestados.append(libro)
            print(f"{self.nombre} ha tomado prestado '{libro.titulo}'.")
        else:
            print(f"El libro '{libro.titulo}' ya está prestado.")

    def devolver_libro(self, libro):
        if libro in self.libros_prestados and libro.devolver():
            self.libros_prestados.remove(libro)
            print(f"{self.nombre} ha devuelto '{libro.titulo}'.")
        else:
            print(f"No puedes devolver '{libro.titulo}', no lo tienes prestado o no se pudo devolver.")

    def mostrar_libros(self):
        if self.libros_prestados:
            print(f"\n{self.nombre} tiene los siguientes libros prestados:")
            for libro in self.libros_prestados:
                print(f"- {libro.titulo}")
        else:
            print(f"\n{self.nombre} no tiene libros prestados.")

# PRACTICA15 U2: Creación de Objetos e Interacción (Función principal)
def main():
    print("--- Simulación de Biblioteca ---")
    
    # 1. Crear objetos Libro
    libro1 = Libro("1984", "George Orwell")
    libro2 = Libro("El Principito", "Antoine de Saint-Exupéry")
    
    # 2. Crear objetos Estudiante
    estudiante1 = Estudiante("Juan Pérez", "A001")
    estudiante2 = Estudiante("Luisa Gómez", "B002")

    # 3. Interacciones
    estudiante1.tomar_prestado(libro1)
    estudiante2.tomar_prestado(libro2)
    estudiante1.tomar_prestado(libro2) # Intentar tomar un libro ya prestado

    estudiante1.mostrar_libros()
    estudiante2.mostrar_libros()

    # 4. Devolución
    estudiante1.devolver_libro(libro1)
    estudiante1.mostrar_libros()

if __name__ == "__main__":
    main()
