1. Fundamentos de Generación de Contraseñas

La generación segura de contraseñas requiere combinar diferentes tipos de caracteres de manera aleatoria y controlada, asegurando tanto la seguridad como la usabilidad.

# Fundamentos de generación de contraseñas seguras

import random
import string

# 1. Caracteres básicos para contraseñas
def obtener_caracteres_basicos():
    """Define los conjuntos básicos de caracteres."""

    print("=== CONJUNTOS DE CARACTERES ===")

    # Definir conjuntos de caracteres
    minusculas = string.ascii_lowercase  # 'abcdefghijklmnopqrstuvwxyz'
    mayusculas = string.ascii_uppercase  # 'ABCDEFGHIJKLMNOPQRSTUVWXYZ'
    numeros = string.digits             # '0123456789'
    simbolos = "!@#$%^&*()_+-=[]{}|;:,.<>?"  # Símbolos comunes

    print("Caracteres disponibles:")
    print(f"Minúsculas ({len(minusculas)}): {minusculas}")
    print(f"Mayúsculas ({len(mayusculas)}): {mayusculas}")
    print(f"Números ({len(numeros)}): {numeros}")
    print(f"Símbolos ({len(simbolos)}): {simbolos}")

    # Combinar según criterios
    caracteres_seleccionados = ""

    incluir_minusculas = True
    incluir_mayusculas = True
    incluir_numeros = True
    incluir_simbolos = True

    if incluir_minusculas:
        caracteres_seleccionados += minusculas
    if incluir_mayusculas:
        caracteres_seleccionados += mayusculas
    if incluir_numeros:
        caracteres_seleccionados += numeros
    if incluir_simbolos:
        caracteres_seleccionados += simbolos

    print(f"\nCaracteres seleccionados ({len(caracteres_seleccionados)}): {caracteres_seleccionados}")

    return caracteres_seleccionados

# 2. Algoritmos de generación
def algoritmos_generacion():
    """Demuestra diferentes algoritmos de generación."""

    print("\n=== ALGORITMOS DE GENERACIÓN ===")

    caracteres = obtener_caracteres_basicos()

    # Algoritmo 1: Selección aleatoria simple
    def generar_simple(longitud, caracteres):
        """Genera contraseña con selección aleatoria simple."""
        contraseña = ""
        for _ in range(longitud):
            contraseña += random.choice(caracteres)
        return contraseña

    # Algoritmo 2: Mezcla controlada (asegura diversidad)
    def generar_mezcla(longitud, caracteres):
        """Genera contraseña asegurando diversidad de caracteres."""

        # Separar por tipos
        tipos = {
            'minuscula': string.ascii_lowercase,
            'mayuscula': string.ascii_uppercase,
            'numero': string.digits,
            'simbolo': "!@#$%^&*"
        }

        contraseña = []

        # Asegurar al menos uno de cada tipo (si longitud lo permite)
        tipos_seleccionados = [tipo for tipo in tipos.keys() if tipo in caracteres.lower() or tipo == 'numero']

        posiciones_usadas = 0

        # Colocar al menos uno de cada tipo requerido
        for tipo in tipos_seleccionados:
            if posiciones_usadas < longitud:
                if tipo == 'minuscula':
                    contraseña.append(random.choice(tipos['minuscula']))
                elif tipo == 'mayuscula':
                    contraseña.append(random.choice(tipos['mayuscula']))
                elif tipo == 'numero':
                    contraseña.append(random.choice(tipos['numero']))
                elif tipo == 'simbolo':
                    contraseña.append(random.choice(tipos['simbolo']))
                posiciones_usadas += 1

        # Completar con caracteres aleatorios
        while posiciones_usadas < longitud:
            contraseña.append(random.choice(caracteres))
            posiciones_usadas += 1

        # Mezclar la contraseña
        random.shuffle(contraseña)

        return ''.join(contraseña)

    # Algoritmo 3: Patrón predecible (menos seguro pero fácil de recordar)
    def generar_patron(longitud, caracteres):
        """Genera contraseña con patrón predecible."""

        patron = []

        # Crear patrón: vocal-consonante-número-símbolo
        ciclo = ['minuscula', 'mayuscula', 'numero', 'simbolo']
        posicion = 0

        for i in range(longitud):
            tipo = ciclo[posicion % len(ciclo)]
            posicion += 1

            if tipo == 'minuscula':
                patron.append(random.choice(string.ascii_lowercase))
            elif tipo == 'mayuscula':
                patron.append(random.choice(string.ascii_uppercase))
            elif tipo == 'numero':
                patron.append(random.choice(string.digits))
            elif tipo == 'simbolo':
                patron.append(random.choice("!@#$"))

        return ''.join(patron)

    # Demostrar algoritmos
    longitud = 12

    print(f"\nGenerando contraseñas de longitud {longitud}:")

    contraseña_simple = generar_simple(longitud, caracteres)
    print(f"Simple: {contraseña_simple}")

    contraseña_mezcla = generar_mezcla(longitud, caracteres)
    print(f"Mezcla: {contraseña_mezcla}")

    contraseña_patron = generar_patron(longitud, caracteres)
    print(f"Patrón: {contraseña_patron}")

# 3. Evaluación de seguridad
def evaluacion_seguridad():
    """Evalúa la seguridad de las contraseñas generadas."""

    print("\n=== EVALUACIÓN DE SEGURIDAD ===")

    def evaluar_contraseña(contraseña):
        """Evalúa la seguridad de una contraseña."""

        puntuacion = 0
        comentarios = []

        # Criterios de evaluación
        longitud = len(contraseña)

        # Longitud
        if longitud >= 8:
            puntuacion += 1
            comentarios.append("✅ Longitud adecuada (8+ caracteres)")
        elif longitud >= 6:
            comentarios.append("⚠️ Longitud mínima aceptable")
        else:
            comentarios.append("❌ Muy corta")

        # Variedad de caracteres
        tiene_minuscula = any(c.islower() for c in contraseña)
        tiene_mayuscula = any(c.isupper() for c in contraseña)
        tiene_numero = any(c.isdigit() for c in contraseña)
        tiene_simbolo = any(not c.isalnum() for c in contraseña)

        tipos_caracteres = sum([tiene_minuscula, tiene_mayuscula, tiene_numero, tiene_simbolo])

        if tipos_caracteres >= 4:
            puntuacion += 2
            comentarios.append("✅ Excelente variedad de caracteres")
        elif tipos_caracteres >= 3:
            puntuacion += 1
            comentarios.append("✅ Buena variedad de caracteres")
        elif tipos_caracteres >= 2:
            comentarios.append("⚠️ Variedad limitada")
        else:
            comentarios.append("❌ Muy pocos tipos de caracteres")

        # Patrones comunes (penalización)
        patrones_comunes = ['123', 'abc', 'qwe', 'aaa', '111']
        tiene_patron_comun = any(patron in contraseña.lower() for patron in patrones_comunes)

        if tiene_patron_comun:
            puntuacion -= 1
            comentarios.append("❌ Contiene patrón común (menos seguro)")

        # Entropía aproximada (medida de aleatoriedad)
        entropia = longitud * (tipos_caracteres ** 0.5)
        if entropia > 50:
            puntuacion += 1
            comentarios.append("✅ Alta entropía (muy aleatoria)")
        elif entropia > 30:
            comentarios.append("✅ Entropía aceptable")
        else:
            comentarios.append("⚠️ Baja entropía")

        # Clasificación final
        if puntuacion >= 4:
            nivel = "MUY FUERTE"
            clase = "strength-strong"
        elif puntuacion >= 3:
            nivel = "FUERTE"
            clase = "strength-good"
        elif puntuacion >= 2:
            nivel = "ACEPTABLE"
            clase = "strength-fair"
        else:
            nivel = "DÉBIL"
            clase = "strength-weak"

        return {
            'nivel': nivel,
            'puntuacion': puntuacion,
            'comentarios': comentarios,
            'clase_css': clase
        }

    # Evaluar las contraseñas generadas anteriormente
    contraseñas_ejemplo = [
        "password123",
        "P@ssw0rd2024!",
        "Tr3sC@racolores",
        "abcdefghijklmnop",
        "A1B2C3D4E5F6G7H8I9J0!",
        "x7$mN2pL9qR4tV8wY3zE6"
    ]

    print("Evaluación de contraseñas de ejemplo:")
    print("-" * 60)

    for contraseña in contraseñas_ejemplo:
        evaluacion = evaluar_contraseña(contraseña)

        print(f"\nContraseña: {contraseña}")
        print(f"Nivel: {evaluacion['nivel']} (Puntuación: {evaluacion['puntuacion']})")
        for comentario in evaluacion['comentarios']:
            print(f"  {comentario}")

# Ejecutar demostraciones
if __name__ == "__main__":
    print("🔐 FUNDAMENTOS DE GENERACIÓN DE CONTRASEÑAS")

    obtener_caracteres_basicos()
    algoritmos_generacion()
    evaluacion_seguridad()

    print("\n✅ ¡Demostración completada!")

2. Validación y Criterios de Seguridad

Una contraseña segura debe cumplir múltiples criterios que evalúan su resistencia contra diferentes tipos de ataques. Implementaremos un sistema completo de validación.

# Sistema avanzado de validación de contraseñas

import re
import math

class ValidadorContrasenas:
    """Sistema completo para validar contraseñas."""

    def __init__(self):
        # Patrones peligrosos
        self.patrones_peligrosos = [
            r'(.)\1{2,}',  # Caracteres repetidos (aaa, 111)
            r'123|234|345|456|567|678|789|890',  # Secuencias numéricas
            r'abc|bcd|cde|def|efg|fgh|ghi|hij|ijk|jkl|klm|lmn|mno|nop|opq|pqr|qrs|rst|stu|tuv|uvw|vwx|wxy|xyz',  # Secuencias alfabéticas
            r'qwe|wer|ert|rty|tyu|yui|uio|iop|opa|pas|sdf|dfg|fgh|ghj|hjk|jkl|klñ',  # Patrones de teclado
            r'admin|password|login|user|root|guest|test|demo',  # Palabras comunes
        ]

        # Caracteres similares (confusos)
        self.similares = {
            'i': ['l', '1', 'L'],
            'l': ['i', '1', 'I'],
            '1': ['i', 'l', 'I', 'L'],
            'o': ['0', 'O'],
            '0': ['o', 'O'],
            'I': ['l', '1', 'i'],
            'L': ['1', 'i', 'l']
        }

    def validar_contraseña(self, contraseña):
        """Valida una contraseña contra múltiples criterios."""

        resultado = {
            'longitud': self._validar_longitud(contraseña),
            'variedad': self._validar_variedad(contraseña),
            'patrones': self._validar_patrones(contraseña),
            'similares': self._validar_similares(contraseña),
            'entropia': self._calcular_entropia(contraseña),
            'tiempo_crackeo': self._estimar_tiempo_crackeo(contraseña)
        }

        # Calcular puntuación general
        puntuacion = sum([
            resultado['longitud']['puntuacion'],
            resultado['variedad']['puntuacion'],
            resultado['patrones']['puntuacion'],
            resultado['similares']['puntuacion']
        ])

        resultado['puntuacion_total'] = puntuacion
        resultado['nivel_seguridad'] = self._obtener_nivel_seguridad(puntuacion)

        return resultado

    def _validar_longitud(self, contraseña):
        """Valida la longitud de la contraseña."""

        longitud = len(contraseña)

        if longitud >= 16:
            return {'estado': 'excelente', 'puntuacion': 3, 'mensaje': 'Longitud excelente'}
        elif longitud >= 12:
            return {'estado': 'buena', 'puntuacion': 2, 'mensaje': 'Longitud buena'}
        elif longitud >= 8:
            return {'estado': 'aceptable', 'puntuacion': 1, 'mensaje': 'Longitud mínima aceptable'}
        else:
            return {'estado': 'insuficiente', 'puntuacion': 0, 'mensaje': 'Muy corta'}

    def _validar_variedad(self, contraseña):
        """Valida la variedad de tipos de caracteres."""

        tipos = {
            'minusculas': bool(re.search(r'[a-z]', contraseña)),
            'mayusculas': bool(re.search(r'[A-Z]', contraseña)),
            'numeros': bool(re.search(r'[0-9]', contraseña)),
            'simbolos': bool(re.search(r'[^a-zA-Z0-9]', contraseña))
        }

        tipos_presentes = sum(tipos.values())

        mensajes = []
        if tipos['minusculas']:
            mensajes.append('minúsculas')
        if tipos['mayusculas']:
            mensajes.append('mayúsculas')
        if tipos['numeros']:
            mensajes.append('números')
        if tipos['simbolos']:
            mensajes.append('símbolos')

        if tipos_presentes >= 4:
            return {'estado': 'excelente', 'puntuacion': 3, 'mensaje': f'Todos los tipos: {", ".join(mensajes)}'}
        elif tipos_presentes == 3:
            return {'estado': 'buena', 'puntuacion': 2, 'mensaje': f'Tres tipos: {", ".join(mensajes)}'}
        elif tipos_presentes == 2:
            return {'estado': 'aceptable', 'puntuacion': 1, 'mensaje': f'Dos tipos: {", ".join(mensajes)}'}
        else:
            return {'estado': 'insuficiente', 'puntuacion': 0, 'mensaje': 'Pocos tipos de caracteres'}

    def _validar_patrones(self, contraseña):
        """Busca patrones peligrosos en la contraseña."""

        encontrados = []

        for patron in self.patrones_peligrosos:
            if re.search(patron, contraseña, re.IGNORECASE):
                encontrados.append(patron)

        if not encontrados:
            return {'estado': 'bueno', 'puntuacion': 2, 'mensaje': 'No se encontraron patrones peligrosos'}
        else:
            return {'estado': 'riesgoso', 'puntuacion': 0, 'mensaje': f'Patrones encontrados: {", ".join(encontrados)}'}

    def _validar_similares(self, contraseña):
        """Verifica caracteres similares que pueden confundirse."""

        encontrados = []

        for char in contraseña:
            if char in self.similares:
                similares = self.similares[char]
                for similar in similares:
                    if similar in contraseña:
                        encontrados.append(f'{char} ↔ {similar}')

        if not encontrados:
            return {'estado': 'bueno', 'puntuacion': 1, 'mensaje': 'No hay caracteres similares'}
        else:
            return {'estado': 'precaucion', 'puntuacion': 0, 'mensaje': f'Caracteres similares: {", ".join(set(encontrados))}'}

    def _calcular_entropia(self, contraseña):
        """Calcula la entropía (aleatoriedad) de la contraseña."""

        # Conjunto único de caracteres usados
        caracteres_unicos = len(set(contraseña))

        # Entropía en bits: log2(caracteres_unicos ^ longitud)
        if caracteres_unicos > 0:
            entropia = len(contraseña) * math.log2(caracteres_unicos)
        else:
            entropia = 0

        return {
            'valor': entropia,
            'estado': 'alta' if entropia > 60 else 'media' if entropia > 40 else 'baja',
            'mensaje': f'Entropía: {entropia".1f"} bits'
        }

    def _estimar_tiempo_crackeo(self, contraseña):
        """Estima el tiempo necesario para crackear la contraseña."""

        entropia = self._calcular_entropia(contraseña)['valor']

        # Suposiciones:
        # - Ataque de fuerza bruta: 1000 intentos por segundo
        # - Ataque con diccionario: 10^6 entradas, 100 intentos/segundo

        tiempo_fuerza_bruta = 2 ** entropia / 1000  # segundos
        tiempo_diccionario = 10**6 / 100  # segundos

        tiempo_realista = min(tiempo_fuerza_bruta, tiempo_diccionario)

        if tiempo_realista > 31536000 * 1000:  # Más de 1000 años
            return {'tiempo': 'Siglos', 'estado': 'imposible', 'mensaje': 'Prácticamente imposible de crackear'}
        elif tiempo_realista > 31536000:  # Más de 1 año
            return {'tiempo': 'Años', 'estado': 'muy_seguro', 'mensaje': f'{tiempo_realista/31536000".0f"} años'}
        elif tiempo_realista > 86400:  # Más de 1 día
            return {'tiempo': 'Días', 'estado': 'seguro', 'mensaje': f'{tiempo_realista/86400".0f"} días'}
        elif tiempo_realista > 3600:  # Más de 1 hora
            return {'tiempo': 'Horas', 'estado': 'moderado', 'mensaje': f'{tiempo_realista/3600".0f"} horas'}
        else:
            return {'tiempo': 'Minutos', 'estado': 'inseguro', 'mensaje': f'{tiempo_realista/60".0f"} minutos'}

    def _obtener_nivel_seguridad(self, puntuacion):
        """Convierte puntuación numérica a nivel de seguridad."""

        if puntuacion >= 8:
            return {'nivel': 'EXCELENTE', 'clase': 'strength-strong', 'color': '#00cec9'}
        elif puntuacion >= 6:
            return {'nivel': 'MUY BUENO', 'clase': 'strength-good', 'color': '#00b894'}
        elif puntuacion >= 4:
            return {'nivel': 'BUENO', 'clase': 'strength-fair', 'color': '#fdcb6e'}
        elif puntuacion >= 2:
            return {'nivel': 'REGULAR', 'clase': 'strength-weak', 'color': '#e17055'}
        else:
            return {'nivel': 'DÉBIL', 'clase': 'strength-weak', 'color': '#d63031'}

# Módulo principal del generador de contraseñas

import random
import string
import json
import os
from datetime import datetime

class GeneradorContrasenas:
    """Generador avanzado de contraseñas seguras."""

    def __init__(self):
        self.caracteres_base = {
            'minusculas': string.ascii_lowercase,
            'mayusculas': string.ascii_uppercase,
            'numeros': string.digits,
            'simbolos': "!@#$%^&*()_+-=[]{}|;:,.<>?/"
        }

    def generar_contraseña(self, longitud=12, opciones=None):
        """Genera una contraseña segura según las opciones especificadas."""

        if opciones is None:
            opciones = {
                'minusculas': True,
                'mayusculas': True,
                'numeros': True,
                'simbolos': True,
                'excluir_similares': True,
                'excluir_ambiguos': True
            }

        # Construir conjunto de caracteres
        caracteres = ""

        if opciones.get('minusculas', True):
            caracteres += self.caracteres_base['minusculas']
        if opciones.get('mayusculas', True):
            caracteres += self.caracteres_base['mayusculas']
        if opciones.get('numeros', True):
            caracteres += self.caracteres_base['numeros']
        if opciones.get('simbolos', True):
            caracteres += self.caracteres_base['simbolos']

        # Excluir caracteres similares si está habilitado
        if opciones.get('excluir_similares', True):
            caracteres = self._excluir_similares(caracteres)

        # Excluir caracteres ambiguos si está habilitado
        if opciones.get('excluir_ambiguos', True):
            caracteres = self._excluir_ambiguos(caracteres)

        if not caracteres:
            raise ValueError("Debe seleccionar al menos un tipo de carácter")

        # Generar contraseña asegurando diversidad
        contraseña = self._generar_con_diversidad(longitud, caracteres, opciones)

        return contraseña

    def _excluir_similares(self, caracteres):
        """Excluye caracteres que pueden confundirse visualmente."""

        similares = "il1Lo0O"
        return ''.join(c for c in caracteres if c not in similares)

    def _excluir_ambiguos(self, caracteres):
        """Excluye caracteres ambiguos o problemáticos."""

        ambiguos = "{}[]()|\\'\"`,;.:"
        return ''.join(c for c in caracteres if c not in ambiguos)

    def _generar_con_diversidad(self, longitud, caracteres, opciones):
        """Genera contraseña asegurando diversidad de caracteres."""

        contraseña = []

        # Asegurar al menos uno de cada tipo seleccionado
        tipos_seleccionados = []

        if opciones.get('minusculas', True):
            tipos_seleccionados.append('minusculas')
        if opciones.get('mayusculas', True):
            tipos_seleccionados.append('mayusculas')
        if opciones.get('numeros', True):
            tipos_seleccionados.append('numeros')
        if opciones.get('simbolos', True):
            tipos_seleccionados.append('simbolos')

        # Colocar al menos uno de cada tipo
        posiciones_usadas = 0

        for tipo in tipos_seleccionados:
            if posiciones_usadas < longitud:
                if tipo == 'minusculas':
                    contraseña.append(random.choice(self.caracteres_base['minusculas']))
                elif tipo == 'mayusculas':
                    contraseña.append(random.choice(self.caracteres_base['mayusculas']))
                elif tipo == 'numeros':
                    contraseña.append(random.choice(self.caracteres_base['numeros']))
                elif tipo == 'simbolos':
                    simbolos_seguros = "!@#$%^&*"
                    contraseña.append(random.choice(simbolos_seguros))

                posiciones_usadas += 1

        # Completar con caracteres aleatorios
        while posiciones_usadas < longitud:
            contraseña.append(random.choice(caracteres))
            posiciones_usadas += 1

        # Mezclar la contraseña
        random.shuffle(contraseña)

        return ''.join(contraseña)

class ValidadorContrasenas:
    """Sistema avanzado de validación de contraseñas."""

    def __init__(self):
        self.patrones_peligrosos = [
            r'(.)\1{2,}',  # Repeticiones
            r'123|234|345|456|567|678|789|890',  # Secuencias numéricas
            r'abc|bcd|cde|def|efg|fgh|ghi|hij|ijk|jkl|klm|lmn|mno|nop|opq|pqr|qrs|rst|stu|tuv|uvw|vwx|wxy|xyz',  # Secuencias alfabéticas
            r'qwe|wer|ert|rty|tyu|yui|uio|iop|opa|pas|sdf|dfg|fgh|ghj|hjk|jkl',  # Patrones teclado
            r'admin|password|login|user|root|guest|test|demo',  # Palabras comunes
        ]

    def validar_contraseña(self, contraseña):
        """Valida completamente una contraseña."""

        resultado = {
            'longitud': self._validar_longitud(contraseña),
            'variedad': self._validar_variedad(contraseña),
            'patrones': self._validar_patrones(contraseña),
            'entropia': self._calcular_entropia(contraseña),
            'estimacion_crackeo': self._estimar_tiempo_crackeo(contraseña)
        }

        # Calcular puntuación general (0-10)
        puntuacion = (
            resultado['longitud']['puntuacion'] +
            resultado['variedad']['puntuacion'] +
            resultado['patrones']['puntuacion']
        )

        resultado['puntuacion_total'] = min(puntuacion, 10)
        resultado['nivel_seguridad'] = self._obtener_nivel_seguridad(puntuacion)

        return resultado

    def _validar_longitud(self, contraseña):
        """Valida la longitud."""

        longitud = len(contraseña)

        if longitud >= 16:
            return {'estado': 'excelente', 'puntuacion': 3, 'mensaje': 'Longitud excelente (16+)'}
        elif longitud >= 12:
            return {'estado': 'buena', 'puntuacion': 2, 'mensaje': 'Longitud buena (12+)'}
        elif longitud >= 8:
            return {'estado': 'aceptable', 'puntuacion': 1, 'mensaje': 'Longitud mínima (8+)'}
        else:
            return {'estado': 'insuficiente', 'puntuacion': 0, 'mensaje': 'Muy corta (<8)'}

    def _validar_variedad(self, contraseña):
        """Valida la variedad de caracteres."""

        import re

        tipos = {
            'minusculas': bool(re.search(r'[a-z]', contraseña)),
            'mayusculas': bool(re.search(r'[A-Z]', contraseña)),
            'numeros': bool(re.search(r'[0-9]', contraseña)),
            'simbolos': bool(re.search(r'[^a-zA-Z0-9]', contraseña))
        }

        tipos_presentes = sum(tipos.values())

        if tipos_presentes >= 4:
            return {'estado': 'excelente', 'puntuacion': 3, 'mensaje': 'Máxima variedad'}
        elif tipos_presentes == 3:
            return {'estado': 'buena', 'puntuacion': 2, 'mensaje': 'Buena variedad'}
        elif tipos_presentes == 2:
            return {'estado': 'aceptable', 'puntuacion': 1, 'mensaje': 'Variedad limitada'}
        else:
            return {'estado': 'insuficiente', 'puntuacion': 0, 'mensaje': 'Poca variedad'}

    def _validar_patrones(self, contraseña):
        """Busca patrones peligrosos."""

        import re

        encontrados = []
        for patron in self.patrones_peligrosos:
            if re.search(patron, contraseña, re.IGNORECASE):
                encontrados.append(patron.pattern)

        if not encontrados:
            return {'estado': 'bueno', 'puntuacion': 2, 'mensaje': 'Sin patrones peligrosos'}
        else:
            return {'estado': 'riesgoso', 'puntuacion': 0, 'mensaje': f'Patrones: {", ".join(encontrados)}'}

    def _calcular_entropia(self, contraseña):
        """Calcula la entropía de la contraseña."""

        caracteres_unicos = len(set(contraseña))

        if caracteres_unicos > 0:
            entropia = len(contraseña) * (caracteres_unicos ** 0.5)
        else:
            entropia = 0

        return {
            'valor': entropia,
            'estado': 'alta' if entropia > 60 else 'media' if entropia > 40 else 'baja',
            'mensaje': f'Entropía: {entropia".1f"} bits'
        }

    def _estimar_tiempo_crackeo(self, contraseña):
        """Estima tiempo de crackeo."""

        entropia = self._calcular_entropia(contraseña)['valor']

        tiempo_segundos = 2 ** entropia / 1000000  # Asumiendo 1M intentos/segundo

        if tiempo_segundos > 31536000 * 1000:  # Más de 1000 años
            return {'tiempo': 'Siglos', 'estado': 'imposible', 'mensaje': 'Imposible de crackear'}
        elif tiempo_segundos > 31536000:  # Más de 1 año
            return {'tiempo': 'Años', 'estado': 'muy_seguro', 'mensaje': f'{tiempo_segundos/31536000".0f"} años'}
        elif tiempo_segundos > 86400:  # Más de 1 día
            return {'tiempo': 'Días', 'estado': 'seguro', 'mensaje': f'{tiempo_segundos/86400".0f"} días'}
        elif tiempo_segundos > 3600:  # Más de 1 hora
            return {'tiempo': 'Horas', 'estado': 'moderado', 'mensaje': f'{tiempo_segundos/3600".0f"} horas'}
        else:
            return {'tiempo': 'Minutos', 'estado': 'inseguro', 'mensaje': f'{tiempo_segundos/60".0f"} minutos'}

    def _obtener_nivel_seguridad(self, puntuacion):
        """Obtiene el nivel de seguridad basado en puntuación."""

        if puntuacion >= 8:
            return {'nivel': 'EXCELENTE', 'clase': 'strength-strong', 'color': '#00cec9'}
        elif puntuacion >= 6:
            return {'nivel': 'MUY BUENO', 'clase': 'strength-good', 'color': '#00b894'}
        elif puntuacion >= 4:
            return {'nivel': 'BUENO', 'clase': 'strength-fair', 'color': '#fdcb6e'}
        elif puntuacion >= 2:
            return {'nivel': 'REGULAR', 'clase': 'strength-weak', 'color': '#e17055'}
        else:
            return {'nivel': 'DÉBIL', 'clase': 'strength-weak', 'color': '#d63031'}

# Módulo para gestión del historial de contraseñas

import json
import os
from datetime import datetime
from typing import List, Dict, Optional

class GestorHistorial:
    """Gestor del historial de contraseñas generadas."""

    def __init__(self, archivo_historial="password_history.json"):
        self.archivo_historial = archivo_historial
        self._asegurar_archivo()

    def _asegurar_archivo(self):
        """Asegura que el archivo de historial existe."""

        if not os.path.exists(self.archivo_historial):
            os.makedirs(os.path.dirname(self.archivo_historial), exist_ok=True)
            with open(self.archivo_historial, 'w', encoding='utf-8') as f:
                json.dump({"historial": [], "configuracion": {}}, f, indent=2)

    def agregar_contraseña(self, contraseña: str, metadata: Optional[Dict] = None):
        """Agrega una contraseña al historial."""

        # Crear entrada de historial
        entrada = {
            "contraseña": contraseña,
            "fecha_generacion": datetime.now().isoformat(),
            "longitud": len(contraseña),
            "metadata": metadata or {}
        }

        try:
            # Leer historial actual
            with open(self.archivo_historial, 'r', encoding='utf-8') as f:
                datos = json.load(f)

            # Agregar nueva entrada
            datos["historial"].append(entrada)

            # Limitar tamaño del historial (últimas 1000 entradas)
            if len(datos["historial"]) > 1000:
                datos["historial"] = datos["historial"][-1000:]

            # Guardar historial actualizado
            with open(self.archivo_historial, 'w', encoding='utf-8') as f:
                json.dump(datos, f, indent=2, ensure_ascii=False)

            return True

        except Exception as e:
            print(f"❌ Error guardando en historial: {e}")
            return False

    def obtener_historial(self, limite: Optional[int] = None) -> List[Dict]:
        """Obtiene el historial de contraseñas."""

        try:
            with open(self.archivo_historial, 'r', encoding='utf-8') as f:
                datos = json.load(f)

            historial = datos.get("historial", [])

            if limite:
                historial = historial[-limite:]

            return historial

        except Exception as e:
            print(f"❌ Error leyendo historial: {e}")
            return []

    def buscar_contraseñas(self, criterio: str) -> List[Dict]:
        """Busca contraseñas que contengan el criterio."""

        historial = self.obtener_historial()

        return [
            entrada for entrada in historial
            if criterio.lower() in entrada["contraseña"].lower()
        ]

    def limpiar_historial(self, dias_antiguos: int = 30) -> bool:
        """Limpia entradas más antiguas que los días especificados."""

        try:
            from datetime import timedelta

            fecha_limite = datetime.now() - timedelta(days=dias_antiguos)

            with open(self.archivo_historial, 'r', encoding='utf-8') as f:
                datos = json.load(f)

            historial_original = datos.get("historial", [])
            historial_filtrado = []

            for entrada in historial_original:
                fecha_entrada = datetime.fromisoformat(entrada["fecha_generacion"])

                if fecha_entrada > fecha_limite:
                    historial_filtrado.append(entrada)

            datos["historial"] = historial_filtrado

            with open(self.archivo_historial, 'w', encoding='utf-8') as f:
                json.dump(datos, f, indent=2, ensure_ascii=False)

            entradas_eliminadas = len(historial_original) - len(historial_filtrado)
            print(f"✅ Historial limpiado: {entradas_eliminadas} entradas eliminadas")

            return True

        except Exception as e:
            print(f"❌ Error limpiando historial: {e}")
            return False

    def obtener_estadisticas(self) -> Dict:
        """Obtiene estadísticas del historial."""

        historial = self.obtener_historial()

        if not historial:
            return {"total_contraseñas": 0, "longitud_promedio": 0}

        longitudes = [entrada["longitud"] for entrada in historial]

        return {
            "total_contraseñas": len(historial),
            "longitud_promedio": sum(longitudes) / len(longitudes),
            "longitud_maxima": max(longitudes),
            "longitud_minima": min(longitudes),
            "fecha_primera": historial[0]["fecha_generacion"] if historial else None,
            "fecha_ultima": historial[-1]["fecha_generacion"] if historial else None
        }

    def exportar_historial(self, formato: str = "json") -> Optional[str]:
        """Exporta el historial en el formato especificado."""

        historial = self.obtener_historial()

        if formato.lower() == "json":
            return json.dumps(historial, indent=2, ensure_ascii=False)

        elif formato.lower() == "csv":
            if not historial:
                return None

            csv_data = "Fecha,Contraseña,Longitud\n"

            for entrada in historial:
                fecha = entrada["fecha_generacion"]
                contraseña = entrada["contraseña"]
                longitud = entrada["longitud"]

                csv_data += f"{fecha},{contraseña},{longitud}\n"

            return csv_data

        elif formato.lower() == "txt":
            if not historial:
                return None

            txt_data = "HISTORIAL DE CONTRASEÑAS\n"
            txt_data += "=" * 50 + "\n\n"

            for entrada in historial:
                txt_data += f"Fecha: {entrada['fecha_generacion']}\n"
                txt_data += f"Contraseña: {entrada['contraseña']}\n"
                txt_data += f"Longitud: {entrada['longitud']}\n"
                txt_data += "-" * 30 + "\n"

            return txt_data

        else:
            raise ValueError(f"Formato no soportado: {formato}")
