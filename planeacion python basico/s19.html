Objetivo de la Sesión

En esta sesión, explorarás uno de los conceptos más importantes y a menudo confusos en programación: el alcance de las variables (variable scope). Comprenderás cómo Python determina dónde una variable es visible y cómo puedes controlar su accesibilidad en diferentes partes de tu programa.

Las variables locales solo son visibles dentro de su función, mientras que las variables globales pueden ser accedidas desde cualquier parte del programa.

Conceptos Clave

1. Variables Locales vs Globales

Las variables en Python tienen diferentes niveles de visibilidad según dónde se definan. Las variables locales solo existen dentro de la función donde se crean, mientras que las variables globales pueden ser accedidas desde cualquier parte del programa.

# Variables globales - definidas fuera de cualquier función
contador_global = 0
mensaje_global = "Hola desde el ámbito global"

def funcion_con_variables_locales():
    """Ejemplo de función con variables locales."""
    # Variables locales - solo existen dentro de esta función
    contador_local = 10
    mensaje_local = "Hola desde el ámbito local"
    resultado = contador_global + contador_local  # Puede acceder a variables globales
    
    print(f"Dentro de la función: {resultado}")
    print(f"Mensaje global: {mensaje_global}")
    print(f"Mensaje local: {mensaje_local}")
    
    return resultado

def otra_funcion():
    """Otra función que intenta acceder a variables."""
    # contador_local no existe aquí - daría error
    # print(contador_local)  # NameError: name 'contador_local' is not defined
    
    # Pero puede acceder a variables globales
    print(f"Mensaje global desde otra función: {mensaje_global}")

# Llamadas a las funciones
resultado = funcion_con_variables_locales()
print(f"Resultado devuelto: {resultado}")

otra_funcion()

# Intento de acceder a variable local fuera de su función (error)
try:
    print(f"Contador local fuera de función: {contador_local}")
except NameError as e:
    print(f"Error esperado: {e}")

# Acceso a variable global fuera de funciones (funciona)
print(f"Contador global fuera de funciones: {contador_global}")

# Ejemplo con funciones anidadas
def funcion_externa():
    """Función externa con variable local."""
    variable_externa = "Soy de la función externa"
    
    def funcion_interna():
        """Función interna que puede acceder a variables de la externa."""
        variable_interna = "Soy de la función interna"
        print(f"Desde interna: {variable_externa}")  # Puede acceder a variable_externa
        print(f"Desde interna: {variable_interna}")
    
    funcion_interna()
    # variable_interna no existe aquí
    # print(variable_interna)  # NameError
    
    print(f"Desde externa: {variable_externa}")

funcion_externa()
Variables Locales
  • Se definen dentro de una función
  • Solo existen mientras la función se ejecuta
  • No pueden ser accedidas desde fuera de la función
  • Se crean cada vez que se llama a la función
  • Tienen prioridad sobre variables globales con el mismo nombre
def funcion():
    local = 42  # Variable local
    print(local)  # Funciona
Variables Globales
  • Se definen fuera de cualquier función
  • Existen durante toda la ejecución del programa
  • Pueden ser accedidas desde cualquier función
  • Se crean una sola vez al inicio del programa
  • Pueden ser "sombras" por variables locales
global = 42  # Variable global

def funcion():
    print(global)  # Puede acceder

Sombreado de Variables

Si defines una variable local con el mismo nombre que una variable global, la variable local "sombrea" (oculta) a la variable global dentro de esa función:

variable = "global"

def funcion():
    variable = "local"  # Sombrea la variable global
    print(variable)     # Imprime "local"

funcion()
print(variable)  # Imprime "global"

2. Modificar Variables Globales desde Funciones

Para modificar una variable global desde dentro de una función, debes usar la palabra clave global. Sin ella, Python creará una nueva variable local con ese nombre.

# Variable global que queremos modificar
contador = 0
lista_global = []

def incrementar_contador():
    """Intenta incrementar el contador global (sin global)."""
    contador = contador + 1  # Crea variable local, no modifica la global
    # contador += 1  # También crea variable local

def incrementar_contador_correcto():
    """Incrementa correctamente el contador global."""
    global contador  # Declara que vamos a usar la variable global
    contador += 1

def agregar_a_lista(elemento):
    """Agrega elemento a lista global."""
    global lista_global
    lista_global.append(elemento)

def crear_contador_personalizado():
    """Crea un contador personalizado usando variables globales."""
    global contador_personalizado
    
    contador_personalizado = 0
    
    def incrementar():
        global contador_personalizado
        contador_personalizado += 1
        return contador_personalizado
    
    def decrementar():
        global contador_personalizado
        contador_personalizado -= 1
        return contador_personalizado
    
    def obtener_valor():
        global contador_personalizado
        return contador_personalizado
    
    def resetear():
        global contador_personalizado
        contador_personalizado = 0
    
    return {
        "incrementar": incrementar,
        "decrementar": decrementar,
        "obtener": obtener_valor,
        "resetear": resetear
    }

# Uso de las funciones
print(f"Contador inicial: {contador}")

incrementar_contador()
print(f"Después de incrementar (incorrecto): {contador}")  # Sigue siendo 0

incrementar_contador_correcto()
print(f"Después de incrementar (correcto): {contador}")  # Ahora es 1

incrementar_contador_correcto()
incrementar_contador_correcto()
print(f"Después de más incrementos: {contador}")  # Ahora es 3

agregar_a_lista("elemento 1")
agregar_a_lista("elemento 2")
print(f"Lista global: {lista_global}")

# Uso del contador personalizado
contador_personalizado = crear_contador_personalizado()

print(f"\nValor inicial: {contador_personalizado['obtener']()}")
contador_personalizado['incrementar']()
contador_personalizado['incrementar']()
print(f"Después de incrementar: {contador_personalizado['obtener']()}")
contador_personalizado['decrementar']()
print(f"Después de decrementar: {contador_personalizado['obtener']()}")
contador_personalizado['resetear']()
print(f"Después de resetear: {contador_personalizado['obtener']()}")

configuracion = {
    "tema": "claro",

def cambiar_configuracion(clave, valor):

Uso Moderado de Variables Globales

El uso excesivo de variables globales puede hacer que el código sea difícil de mantener y depurar. Algunas recomendaciones:

  • Usa variables globales solo cuando sea realmente necesario
  • Considera pasar parámetros y devolver valores en lugar de modificar globales
  • Usa constantes globales (en mayúsculas) para valores que no cambien
  • Documenta claramente qué variables globales usas y por qué

3. Reglas de Alcance y Problemas Comunes

Python sigue reglas específicas para determinar el alcance de las variables. Comprender estas reglas te ayudará a evitar errores comunes y escribir código más predecible.

# Regla 1: Las funciones buscan variables en orden LEGB (Local, Enclosing, Global, Built-in)
def funcion_principal():
    """Función que demuestra el orden de búsqueda LEGB."""
    variable_global = "global"
    constante = 42
    
    def funcion_anidada():
        """Función anidada que busca variables."""
        # 1. Primero busca en ámbito local (no hay)
        # 2. Luego busca en ámbito enclosing (funcion_principal)
        print(f"Variable global desde anidada: {variable_global}")
        print(f"Constante desde anidada: {constante}")
        
        # 3. Si no encuentra, busca en ámbito global
        # 4. Finalmente busca en built-in (funciones como print, len, etc.)
        print(f"Función print desde anidada: {print}")
    
    funcion_anidada()

funcion_principal()

# Regla 2: Las asignaciones crean variables locales
def problema_asignacion():
    """Ejemplo de problema común con asignaciones."""
    # Esta variable existe en el ámbito global
    contador = 10
    
    def modificar_contador():
        """Intenta modificar el contador."""
        # ¡PROBLEMA! Esta línea crea una variable local llamada contador
        contador = contador + 1  # UnboundLocalError: local variable 'contador' referenced before assignment
        return contador
    
    return modificar_contador

# Solución correcta
def solucion_asignacion():
    """Solución correcta usando global."""
    contador = 10
    
    def modificar_contador():
        global contador
        contador += 1
        return contador
    
    return modificar_contador

# Regla 3: Variables en bucles y condicionales
def variables_en_bucles():
    """Variables creadas en bucles y condicionales."""
    for i in range(3):
        variable_en_bucle = i * 2
        print(f"Variable en bucle {i}: {variable_en_bucle}")
    
    # variable_en_bucle existe aquí (aunque no es recomendable usarla)
    print(f"Variable fuera del bucle: {variable_en_bucle}")

# Problema común: variables en listas por comprensión vs bucles
def listas_vs_bucles():
    """Diferencia entre listas por comprensión y bucles."""
    # Lista por comprensión - las variables están en el mismo ámbito
    numeros = [1, 2, 3, 4, 5]
    cuadrados = [n**2 for n in numeros]  # n existe aquí
    
    # Bucle tradicional - las variables están en el mismo ámbito
    for n in numeros:
        cuadrado = n**2
    
    print(f"Último cuadrado calculado: {cuadrado}")  # cuadrado existe aquí

# Ejemplo práctico: contador de funciones
contador_llamadas = 0

def funcion_con_contador():
    """Función que cuenta cuántas veces ha sido llamada."""
    global contador_llamadas
    contador_llamadas += 1
    print(f"Esta función ha sido llamada {contador_llamadas} veces")

def funcion_con_contador_local():
    """Función con contador local (mejor práctica)."""
    contador_llamadas_local = 0
    
    def inner():
        nonlocal contador_llamadas_local
        contador_llamadas_local += 1
        print(f"Llamadas locales: {contador_llamadas_local}")
    
    return inner

# Uso de las funciones
print("=== EJEMPLOS DE ALCANCE ===")
funcion_con_contador()
funcion_con_contador()

contador_local = funcion_con_contador_local()
contador_local()
contador_local()

# Ejemplo avanzado: closure (función que recuerda su entorno)
def crear_contador_inicial(inicial=0):
    """Crea un contador que recuerda su valor inicial."""
    valor_actual = inicial
    
    def contador():
        nonlocal valor_actual
        valor_actual += 1
        return valor_actual
    
    def obtener_valor():
        nonlocal valor_actual
        return valor_actual
    
    def resetear(nuevo_valor=0):
        nonlocal valor_actual
        valor_actual = nuevo_valor
    
    return {
        "incrementar": contador,
        "obtener": obtener_valor,
        "resetear": resetear
    }

# Uso del contador avanzado
contador1 = crear_contador_inicial(10)
contador2 = crear_contador_inicial(100)

print(f"\nContador 1 inicial: {contador1['obtener']()}")
contador1['incrementar']()
print(f"Contador 1 después de incrementar: {contador1['obtener']()}")

print(f"Contador 2 inicial: {contador2['obtener']()}")
contador2['incrementar']()
print(f"Contador 2 después de incrementar: {contador2['obtener']()}")

# Cada contador mantiene su propio estado independiente
contador1['incrementar']()
print(f"Contador 1: {contador1['obtener']()}, Contador 2: {contador2['obtener']()}")

Problemas Comunes

  • UnboundLocalError: Intentar usar una variable antes de asignarla en una función
  • Sombreado accidental: Crear variables locales que ocultan variables globales
  • Variables en listas por comprensión: Las variables existen fuera de la comprensión
  • Modificar listas/diccionarios globales: Puede causar efectos secundarios inesperados

Regla LEGB

Python busca variables en este orden:

  1. Local: Variables definidas en la función actual
  2. Enclosing: Variables en funciones que contienen a la actual
  3. Global: Variables definidas en el módulo principal
  4. Built-in: Funciones y constantes predefinidas de Python

Exploraremos cómo interactúan las variables locales y globales en diferentes situaciones.

Interacción entre Ámbitos

Ejemplo Completo de Alcance
# Variables globales
usuarios_registrados = 0
productos_disponibles = ["laptop", "mouse", "teclado"]
configuracion_sistema = {
    "modo_debug": False,
    "max_usuarios": 100,
    "idioma": "español"
}

def registrar_usuario(nombre_usuario):
    """Registra un nuevo usuario en el sistema."""
    global usuarios_registrados
    
    # Validar que no excedamos el máximo
    if usuarios_registrados >= configuracion_sistema["max_usuarios"]:
        print(f"❌ Error: Se alcanzó el máximo de {configuracion_sistema['max_usuarios']} usuarios")
        return False
    
    # Simular registro
    usuarios_registrados += 1
    
    # Variable local para el proceso de registro
    codigo_verificacion = generar_codigo_aleatorio()
    tiempo_registro = obtener_tiempo_actual()
    
    print(f"✅ Usuario '{nombre_usuario}' registrado exitosamente")
    print(f"📧 Código de verificación: {codigo_verificacion}")
    print(f"🕐 Registrado en: {tiempo_registro}")
    
    return True

def generar_codigo_aleatorio():
    """Genera un código de verificación aleatorio."""
    import random
    return random.randint(100000, 999999)

def obtener_tiempo_actual():
    """Obtiene la fecha y hora actual."""
    from datetime import datetime
    return datetime.now().strftime("%d/%m/%Y %H:%M:%S")

def mostrar_estadisticas():
    """Muestra estadísticas actuales del sistema."""
    print("
=== ESTADÍSTICAS DEL SISTEMA ===")
    print(f"Usuarios registrados: {usuarios_registrados}")
    print(f"Productos disponibles: {len(productos_disponibles)}")
    print(f"Modo debug: {'Activado' if configuracion_sistema['modo_debug'] else 'Desactivado'}")
    print(f"Idioma: {configuracion_sistema['idioma']}")

def cambiar_configuracion(opcion, valor):
    """Cambia una configuración del sistema."""
    global configuracion_sistema
    
    if opcion not in configuracion_sistema:
        print(f"❌ Configuración '{opcion}' no existe")
        return False
    
    valor_anterior = configuracion_sistema[opcion]
    configuracion_sistema[opcion] = valor
    
    print(f"✅ Configuración '{opcion}' cambiada de '{valor_anterior}' a '{valor}'")
    return True

def agregar_producto(producto):
    """Agrega un producto a la lista global."""
    global productos_disponibles
    
    if producto.lower() in [p.lower() for p in productos_disponibles]:
        print(f"⚠️ El producto '{producto}' ya está en la lista")
        return False
    
    productos_disponibles.append(producto)
    print(f"✅ Producto '{producto}' agregado a la lista")
    return True

Ejemplo 2: Programa de Configuración Global

Crearemos un programa que gestione configuración global de manera efectiva.

Sistema de Configuración Avanzado

Sistema de Configuración Completo
# Configuración global del sistema
CONFIGURACION = {
    "base_datos": {
        "host": "localhost",
        "puerto": 5432,
        "usuario": "admin",
        "contrasena": "secreta",
        "base_datos": "sistema_principal"
    },
    "servidor_web": {
        "host": "0.0.0.0",
        "puerto": 8000,
        "debug": False,
        "ssl": False
    },
    "aplicacion": {
        "nombre": "Sistema de Gestión",
        "version": "2.1.0",
        "entorno": "desarrollo",
        "log_level": "INFO"
    },
    "seguridad": {
        "max_intentos_login": 3,
        "tiempo_bloqueo": 300,  # segundos
        "requerir_ssl": False
    }
}

# Variables globales simples
usuarios_conectados = 0
operaciones_realizadas = 0
ultima_operacion = None

def obtener_configuracion(seccion, clave=None):
    """
    Obtiene configuración de una sección específica.
    
    Args:
        seccion (str): Nombre de la sección de configuración
        clave (str): Clave específica dentro de la sección (opcional)
        
    Returns:
        dict or any: Configuración solicitada
    """
    global CONFIGURACION
    
    if seccion not in CONFIGURACION:
        return None
    
    if clave is None:
        return CONFIGURACION[seccion]
    
    return CONFIGURACION[seccion].get(clave)

def actualizar_configuracion(seccion, clave, valor):
    """
    Actualiza una configuración específica.
    
    Args:
        seccion (str): Sección de configuración
        clave (str): Clave a actualizar
        valor: Nuevo valor
        
    Returns:
        bool: True si se actualizó correctamente
    """
    global CONFIGURACION
    
    if seccion not in CONFIGURACION:
        print(f"❌ Sección '{seccion}' no existe")
        return False
    
    if clave not in CONFIGURACION[seccion]:
        print(f"❌ Clave '{clave}' no existe en sección '{seccion}'")
        return False
    
    valor_anterior = CONFIGURACION[seccion][clave]
    CONFIGURACION[seccion][clave] = valor
    
    print(f"✅ Configuración actualizada: {seccion}.{clave} = {valor_anterior} → {valor}")
    return True

def mostrar_configuracion_completa():
    """Muestra toda la configuración del sistema."""
    global CONFIGURACION
    
    print("\n=== CONFIGURACIÓN COMPLETA DEL SISTEMA ===")
    
    for seccion, datos in CONFIGURACION.items():
        print(f"\n[{seccion.upper()}]")
        print("-" * (len(seccion) + 2))
        for clave, valor in datos.items():
            print(f"{clave}: {valor}")

def incrementar_estadisticas(operacion):
    """Incrementa las estadísticas globales."""
    global usuarios_conectados, operaciones_realizadas, ultima_operacion
    
    usuarios_conectados += 1
    operaciones_realizadas += 1
    ultima_operacion = operacion

def mostrar_estadisticas():
    """Muestra estadísticas actuales."""
    global usuarios_conectados, operaciones_realizadas, ultima_operacion
    
    print("\n=== ESTADÍSTICAS DEL SISTEMA ===")
    print(f"Usuarios conectados: {usuarios_conectados}")
    print(f"Operaciones realizadas: {operaciones_realizadas}")
    print(f"Última operación: {ultima_operacion or 'Ninguna'}")

def resetear_estadisticas():
    """Resetea las estadísticas a cero."""
    global usuarios_conectados, operaciones_realizadas, ultima_operacion
    
    usuarios_conectados = 0
    operaciones_realizadas = 0
    ultima_operacion = None
    
    print("✅ Estadísticas reseteadas")

# Funciones específicas del negocio
def conectar_base_datos():
    """Simula conexión a la base de datos."""
    config_db = obtener_configuracion("base_datos")
    
    print(f"🔌 Conectando a {config_db['host']}:{config_db['puerto']}")
    print(f"📊 Base de datos: {config_db['base_datos']}")
    
    incrementar_estadisticas("conexion_db")
    return True

def iniciar_servidor_web():
    """Simula inicio del servidor web."""
    config_web = obtener_configuracion("servidor_web")
    
    debug_status = " (modo debug)" if config_web["debug"] else ""
    print(f"🌐 Iniciando servidor en {config_web['host']}:{config_web['puerto']}{debug_status}")
    
    incrementar_estadisticas("inicio_servidor")
    return True

def validar_acceso(usuario, contrasena):
    """Valida el acceso de un usuario."""
    config_seguridad = obtener_configuracion("seguridad")
    max_intentos = config_seguridad["max_intentos_login"]
    
    # Simulación simple de validación
    acceso_valido = usuario == "admin" and contrasena == "secreta"
    
    if acceso_valido:
        print(f"✅ Acceso concedido a '{usuario}'")
        incrementar_estadisticas(f"login_exitoso_{usuario}")
    else:
        print(f"❌ Acceso denegado. Intentos máximos: {max_intentos}")
        incrementar_estadisticas("login_fallido")
    
    return acceso_valido

# Uso del sistema de configuración
print("=== SISTEMA DE CONFIGURACIÓN AVANZADO ===")

# Mostrar configuración inicial
mostrar_configuracion_completa()
mostrar_estadisticas()

# Simular operaciones del sistema
print("\n--- OPERACIONES DEL SISTEMA ---")
conectar_base_datos()
iniciar_servidor_web()

# Intentos de login

validar_acceso("usuario", "incorrecta")  # Fallo

mostrar_estadisticas()

# Cambiar configuración
print("\n--- CAMBIOS DE CONFIGURACIÓN ---")
actualizar_configuracion("servidor_web", "puerto", 9000)
actualizar_configuracion("aplicacion", "entorno", "produccion")

mostrar_configuracion_completa()

Módulo con Estado Interno

Módulo de Gestión de Inventario
# Simulación de un módulo de gestión de inventario
# Este código podría estar en un archivo separado (inventario.py)

# Estado interno del módulo (variables "globales" del módulo)
_inventario = {}  # Diccionario privado que almacena el inventario
_operaciones = []  # Lista privada que registra operaciones
_contador_productos = 0

def agregar_producto(nombre, cantidad, precio_unitario):
    """
    Agrega un producto al inventario.
    
    Args:
        nombre (str): Nombre del producto
        cantidad (int): Cantidad inicial
        precio_unitario (float): Precio por unidad
        
    Returns:
        bool: True si se agregó correctamente
    """
    global _inventario, _operaciones, _contador_productos
    
    if nombre in _inventario:
        print(f"⚠️ El producto '{nombre}' ya existe. Use actualizar_cantidad() para modificarlo.")
        return False
    
    _inventario[nombre] = {
        "cantidad": cantidad,
        "precio_unitario": precio_unitario,
        "valor_total": cantidad * precio_unitario
    }
    
    _contador_productos += 1
    _operaciones.append({
        "tipo": "agregar",
        "producto": nombre,
        "cantidad": cantidad,
        "timestamp": obtener_timestamp()
    })
    
    print(f"✅ Producto '{nombre}' agregado: {cantidad} unidades a ${precio_unitario} cada una")
    return True

def actualizar_cantidad(nombre, nueva_cantidad):
    """
    Actualiza la cantidad de un producto existente.
    
    Args:
        nombre (str): Nombre del producto
        nueva_cantidad (int): Nueva cantidad
        
    Returns:
        bool: True si se actualizó correctamente
    """
    global _inventario, _operaciones
    
    if nombre not in _inventario:
        print(f"❌ El producto '{nombre}' no existe en el inventario")
        return False
    
    cantidad_anterior = _inventario[nombre]["cantidad"]
    diferencia = nueva_cantidad - cantidad_anterior
    
    _inventario[nombre]["cantidad"] = nueva_cantidad
    _inventario[nombre]["valor_total"] = nueva_cantidad * _inventario[nombre]["precio_unitario"]
    
    _operaciones.append({
        "tipo": "actualizar",
        "producto": nombre,
        "cantidad_anterior": cantidad_anterior,
        "cantidad_nueva": nueva_cantidad,
        "diferencia": diferencia,
        "timestamp": obtener_timestamp()
    })
    
    accion = "incrementada" if diferencia > 0 else "reducida"
    print(f"✅ Cantidad de '{nombre}' {accion} en {abs(diferencia)} unidades")
    return True

def obtener_producto(nombre):
    """
    Obtiene información de un producto específico.
    
    Args:
        nombre (str): Nombre del producto
        
    Returns:
        dict or None: Información del producto o None si no existe
    """
    global _inventario
    return _inventario.get(nombre)

def listar_productos():
    """Lista todos los productos en el inventario."""
    global _inventario, _contador_productos
    
    if not _inventario:
        print("📦 El inventario está vacío")
        return
    
    print(f"\n=== INVENTARIO ACTUAL ({_contador_productos} productos) ===")
    print(f"{'Producto'"<20"} {'Cantidad'"<10"} {'Precio Unit.'" <12" {'Valor Total'"<15"")
    print("-" * 60)
    
    for nombre, datos in _inventario.items():
        print(f"{nombre"<20"} {datos['cantidad']"<10"} ${datos['precio_unitario']"<12.2f"} ${datos['valor_total']"<15.2f"}")

def calcular_valor_total():
    """Calcula el valor total del inventario."""
    global _inventario
    
    valor_total = sum(datos["valor_total"] for datos in _inventario.values())
    print(f"💰 Valor total del inventario: ${valor_total:.2f}")
    return valor_total

def obtener_estadisticas():
    """Obtiene estadísticas del módulo."""
    global _inventario, _operaciones, _contador_productos
    
    return {
        "productos_totales": _contador_productos,
        "operaciones_realizadas": len(_operaciones),
        "productos_con_stock": len([p for p in _inventario.values() if p["cantidad"] > 0]),
        "valor_inventario": sum(datos["valor_total"] for datos in _inventario.values())
    }

def mostrar_operaciones_recientes(limite=5):
    """Muestra las operaciones más recientes."""
    global _operaciones
    
    if not _operaciones:
        print("📋 No hay operaciones registradas")
        return
    
    print(f"\n=== ÚLTIMAS {min(limite, len(_operaciones))} OPERACIONES ===")
    
    operaciones_mostrar = _operaciones[-limite:] if limite > 0 else _operaciones
    
    for op in reversed(operaciones_mostrar):
        tipo = op["tipo"].upper()
        producto = op["producto"]
        timestamp = op["timestamp"]
        
        if op["tipo"] == "agregar":
            cantidad = op["cantidad"]
            print(f"{tipo} - {producto}: {cantidad} unidades - {timestamp}")
        elif op["tipo"] == "actualizar":
            print(f"{tipo} - {producto}: {op['cantidad_anterior']} → {op['cantidad_nueva']} - {timestamp}")

def resetear_inventario():
    """Resetea completamente el inventario (cuidado!)."""
    global _inventario, _operaciones, _contador_productos
    
    confirmacion = input("⚠️ ¿Está seguro de que desea resetear el inventario? (si/no): ").lower()
    
    if confirmacion == "si":
        _inventario.clear()
        _operaciones.clear()
        _contador_productos = 0
        print("✅ Inventario reseteado completamente")
        return True
    else:
        print("❌ Operación cancelada")
        return False

def obtener_timestamp():
    """Obtiene la marca de tiempo actual."""
    from datetime import datetime
    return datetime.now().strftime("%d/%m/%Y %H:%M:%S")

# Función para crear múltiples módulos independientes
def crear_modulo_inventario():
    """
    Crea un módulo de inventario independiente.
    Esto demuestra cómo crear múltiples instancias con estado separado.
    """
    # Cada llamada crea un nuevo estado independiente
    inventario = {}
    operaciones = []
    contador = 0
    
    def agregar_producto_local(nombre, cantidad, precio):
        nonlocal inventario, operaciones, contador
        
        if nombre in inventario:
            return False
        
        inventario[nombre] = {
            "cantidad": cantidad,
            "precio_unitario": precio,
            "valor_total": cantidad * precio
        }
        
        contador += 1
        operaciones.append({
            "tipo": "agregar",
            "producto": nombre,
            "cantidad": cantidad,
            "timestamp": obtener_timestamp()
        })
        
        return True
    
    def listar_productos_local():
        nonlocal inventario, contador
        
        if not inventario:
            return "Inventario vacío"
        
        resultado = [f"Módulo independiente - {contador} productos:"]
        for nombre, datos in inventario.items():
            resultado.append(f"  {nombre}: {datos['cantidad']} unidades (${datos['precio_unitario']:.2f} c/u)")
        
        return "\n".join(resultado)
    
    return {
        "agregar_producto": agregar_producto_local,
        "listar_productos": listar_productos_local
    }

# Uso del módulo de inventario
print("=== MÓDULO DE GESTIÓN DE INVENTARIO ===")

# Agregar productos al inventario principal
agregar_producto("Laptop Gaming", 10, 1200.50)
agregar_producto("Mouse Inalámbrico", 25, 45.99)
agregar_producto("Teclado Mecánico", 15, 89.99)

# Actualizar cantidad
actualizar_cantidad("Laptop Gaming", 8)

# Mostrar información
listar_productos()
calcular_valor_total()
mostrar_operaciones_recientes(3)

# Crear módulos independientes
inventario_tienda1 = crear_modulo_inventario()
inventario_tienda2 = crear_modulo_inventario()

inventario_tienda1["agregar_producto"]("Producto A", 5, 10.0)
inventario_tienda2["agregar_producto"]("Producto B", 3, 15.0)

print(f"\n{inventario_tienda1['listar_productos']()}")
print(f"\n{inventario_tienda2['listar_productos']()}")

# Estadísticas finales
estadisticas = obtener_estadisticas()
print("
=== ESTADÍSTICAS FINALES ===")
for clave, valor in estadisticas.items():
    print(f"{clave.replace('_', ' ').title()}: {valor}")
