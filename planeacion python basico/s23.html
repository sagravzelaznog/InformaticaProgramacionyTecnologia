1. Escritura Segura en Archivos

La escritura segura implica usar el contexto manager 'with' para garantizar que los archivos se cierren correctamente, manejar diferentes modos de escritura y formatear los datos apropiadamente antes de escribirlos.

# Escritura segura en archivos con diferentes técnicas

# 1. Escritura básica segura
def escritura_basica_segura():
    """Demuestra escritura básica segura en archivos."""

    print("=== ESCRITURA BÁSICA SEGURA ===")

    datos_a_escribir = [
        "Primera línea de datos",
        "Segunda línea con números: 42",
        "Tercera línea con caracteres especiales: ñáéíóú",
        "Última línea del archivo"
    ]

    try:
        # Escritura con contexto manager (modo 'w')
        with open('datos_basicos.txt', 'w', encoding='utf-8') as archivo:
            for dato in datos_a_escribir:
                archivo.write(dato + '\n')  # Agregar salto de línea

        print("✅ Archivo creado exitosamente con modo 'w'")

        # Verificar contenido
        with open('datos_basicos.txt', 'r', encoding='utf-8') as archivo:
            contenido = archivo.read()
        print(f"📄 Contenido escrito:\n{contenido}")

        # Agregar más contenido (modo 'a')
        with open('datos_basicos.txt', 'a', encoding='utf-8') as archivo:
            archivo.write("Línea adicional agregada con modo 'a'\n")
            archivo.write(f"Timestamp: {time.ctime()}\n")

        print("✅ Contenido adicional agregado")

        # Ver contenido final
        with open('datos_basicos.txt', 'r', encoding='utf-8') as archivo:
            contenido_final = archivo.read()
        print(f"📄 Contenido final:\n{contenido_final}")

    except Exception as e:
        print(f"❌ Error: {e}")
    finally:
        # Limpiar
        if os.path.exists('datos_basicos.txt'):
            os.remove('datos_basicos.txt')

def metodos_escritura_avanzados():

    # Método 2: print() con argumento file
    with open('metodo2.txt', 'w', encoding='utf-8') as f:
        print("Esta línea fue escrita con print()", file=f)
        print("Número formateado: {:.2f}".format(3.14159), file=f)
        print("Fecha y hora:", time.ctime(), file=f)

    print("✅ Método print(file=): datos formateados escritos")

    # Método 3: Escritura con formateo avanzado
    datos_estructurados = [
        {"nombre": "Ana", "edad": 25, "ciudad": "Madrid"},
        {"nombre": "Carlos", "edad": 30, "ciudad": "Barcelona"},
        {"nombre": "María", "edad": 28, "ciudad": "Valencia"}
    ]

    with open('metodo3.txt', 'w', encoding='utf-8') as f:
        for persona in datos_estructurados:
            linea = "{nombre},{edad},{ciudad}\n".format(**persona)
            f.write(linea)

    print("✅ Método format(): datos estructurados formateados")

    # Método 4: Escritura binaria
    datos_binarios = bytes([65, 66, 67, 68, 69, 10])  # ABCDE\n

    with open('metodo4.bin', 'wb') as f:
        f.write(datos_binarios)

    print("✅ Método binario: datos binarios escritos")

    # Verificar archivos creados
    for archivo in ['metodo1.txt', 'metodo2.txt', 'metodo3.txt', 'metodo4.bin']:
        if os.path.exists(archivo):
            tamaño = os.path.getsize(archivo)
            print(f"✅ {archivo}: {tamaño} bytes")

            if archivo.endswith('.txt'):
                with open(archivo, 'r', encoding='utf-8') as f:
                    contenido = f.read().strip()
                print(f"   Contenido: {contenido}")

    # Limpiar
    for archivo in ['metodo1.txt', 'metodo2.txt', 'metodo3.txt', 'metodo4.bin']:
        if os.path.exists(archivo):
            os.remove(archivo)

# 3. Modos de escritura y append
def modos_escritura_append():
    """Demuestra diferentes modos de escritura y append."""

    print("\n=== MODOS DE ESCRITURA Y APPEND ===")

    # Crear archivo base
    with open('base.txt', 'w', encoding='utf-8') as f:
        f.write("Contenido inicial\n")
        f.write("Línea 2\n")
        f.write("Línea 3\n")

    print("📁 Archivo base creado")

    # Modo append (a) - agregar al final
    print("📝 Modo 'a' (append):")
    with open('base.txt', 'a', encoding='utf-8') as f:
        f.write("Línea agregada con 'a'\n")
        f.write("Otra línea más\n")
        f.write(f"Timestamp: {time.ctime()}\n")

    with open('base.txt', 'r', encoding='utf-8') as f:
        contenido = f.read()
    print(f"Contenido después de append:\n{contenido}")

    # Modo escritura+lectura (w+) - sobreescribe pero permite lectura
    print("📝 Modo 'w+' (escritura+lectura):")
    with open('nuevo.txt', 'w+', encoding='utf-8') as f:
        f.write("Contenido creado con w+\n")
        f.write("Segunda línea\n")

        # Volver al inicio para leer
        f.seek(0)
        contenido = f.read()
        print(f"Contenido leído inmediatamente: {contenido.strip()}")

        # Agregar más al final
        f.seek(0, 2)  # Ir al final
        f.write("Línea agregada al final\n")

    with open('nuevo.txt', 'r', encoding='utf-8') as f:
        contenido_final = f.read()
    print(f"Contenido final:\n{contenido_final}")

    # Modo append+lectura (a+) - permite lectura y escritura al final
    print("📝 Modo 'a+' (append+lectura):")
    with open('base.txt', 'a+', encoding='utf-8') as f:
        # Leer contenido actual
        f.seek(0)
        contenido_actual = f.read()
        print(f"Contenido actual ({len(contenido_actual)} caracteres)")

        # Agregar al final
        f.write("Línea agregada con a+\n")

    with open('base.txt', 'r', encoding='utf-8') as f:
        contenido_completo = f.read()
    print(f"Contenido completo:\n{contenido_completo}")

    # Limpiar
    for archivo in ['base.txt', 'nuevo.txt']:
        if os.path.exists(archivo):
            os.remove(archivo)

def escritura_archivos_grandes():
    """Demuestra técnicas para escribir archivos grandes eficientemente."""

    print("\n=== ESCRITURA DE ARCHIVOS GRANDES ===")

    # Técnica 1: Escritura en bloques
    print("📦 Escritura en bloques:")
    bloque_tamaño = 8192  # 8KB

    with open('archivo_grande.txt', 'w', encoding='utf-8') as f:
        for i in range(1000):
            linea = f"Línea número {i + 1} con datos adicionales para hacerla más larga\n"
            f.write(linea)

            # Flush ocasional para archivos muy grandes
            if i % 100 == 0:
                f.flush()

    tamaño = os.path.getsize('archivo_grande.txt')
    print(f"✅ Archivo creado: {tamaño:,}","ytes")

    # Técnica 2: Buffer personalizado
    print("🗂️ Buffer personalizado:")
    buffer_lineas = []
    tamaño_buffer = 100

    with open('archivo_buffer.txt', 'w', encoding='utf-8') as f:
        for i in range(500):
            linea = f"Línea bufferizada {i + 1}: datos de ejemplo\n"
            buffer_lineas.append(linea)

            if len(buffer_lineas) >= tamaño_buffer:
                f.writelines(buffer_lineas)
                buffer_lineas = []

        # Escribir líneas restantes
        if buffer_lineas:
            f.writelines(buffer_lineas)

    tamaño_buffer = os.path.getsize('archivo_buffer.txt')
    print(f"✅ Archivo con buffer: {tamaño_buffer:,}","ytes")

    # Técnica 3: Escritura asíncrona simulada
    print("⚡ Escritura asíncrona (simulada):")
    import threading

    def escribir_en_hilo(archivo, datos, hilo_id):
        """Simula escritura en hilo separado."""
        with open(archivo, 'a', encoding='utf-8') as f:
            for dato in datos:
                f.write(f"Hilo {hilo_id}: {dato}\n")
                time.sleep(0.001)  # Simular procesamiento

    # Datos para múltiples hilos
    datos_hilo1 = [f"Dato {i}" for i in range(1, 101)]
    datos_hilo2 = [f"Info {i}" for i in range(1, 101)]

    # Crear hilos
    hilo1 = threading.Thread(target=escribir_en_hilo, args=('archivo_hilos.txt', datos_hilo1, 1))
    hilo2 = threading.Thread(target=escribir_en_hilo, args=('archivo_hilos.txt', datos_hilo2, 2))

    # Ejecutar hilos
    hilo1.start()
    hilo2.start()

    # Esperar a que terminen
    hilo1.join()
    hilo2.join()

    print("✅ Escritura multi-hilo completada")

    # Ver resultado
    with open('archivo_hilos.txt', 'r', encoding='utf-8') as f:
        lineas = f.readlines()

    print(f"📊 Total líneas escritas: {len(lineas)}")

    # Técnica 4: Escritura con compresión (simulada)
    print("🗜️ Escritura con compresión:")
    datos_comprimidos = []
    contador_comprimido = 0

    with open('archivo_comprimido.txt', 'w', encoding='utf-8') as f:

            if i % 10 == 0:  # Cada 10 líneas, comprimir referencia

    print(f"✅ Archivo comprimido: {tamaño_comprimido:,}","ytes")

    # Limpiar archivos

# 1. Formateo básico de tipos de datos
def formateo_basico_datos():
    """Demuestra formateo básico de diferentes tipos de datos."""

    print("=== FORMATEO BÁSICO DE DATOS ===")

    # Datos de ejemplo
    datos = {
        'texto': 'Hola Mundo',
        'entero': 42,
        'flotante': 3.14159,
        'booleano': True,
        'lista': [1, 2, 3, 4, 5],
        'diccionario': {'nombre': 'Ana', 'edad': 25},
        'fecha': datetime.now(),
        'nulo': None
    }

    # Método 1: Conversión simple a string
    with open('datos_basicos.txt', 'w', encoding='utf-8') as f:
        f.write("FORMATO BÁSICO DE DATOS\n")
        f.write("=" * 40 + "\n")

        for clave, valor in datos.items():
            # Convertir cada tipo de dato apropiadamente
            if valor is None:
                valor_str = "None"
            elif isinstance(valor, bool):
                valor_str = "True" if valor else "False"
            elif isinstance(valor, (int, float)):
                valor_str = str(valor)
            elif isinstance(valor, (list, dict)):
                valor_str = str(valor)
            elif isinstance(valor, datetime):
                valor_str = valor.strftime('%Y-%m-%d %H:%M:%S')
            else:
                valor_str = str(valor)

            f.write(f"{clave}: {valor_str}\n")

    print("✅ Datos básicos formateados")

    # Método 2: Formateo estructurado
    print("\n📋 Datos formateados:")
    with open('datos_basicos.txt', 'r', encoding='utf-8') as f:
        contenido = f.read()
    print(contenido)

    # Método 3: Formateo tabular
    productos = [
        {"producto": "Laptop", "precio": 999.99, "cantidad": 10},
        {"producto": "Mouse", "precio": 25.50, "cantidad": 50},
        {"producto": "Teclado", "precio": 75.00, "cantidad": 30}
    ]

    with open('productos_tabla.txt', 'w', encoding='utf-8') as f:
        # Encabezado
        f.write(f"{'Producto'"<15"{'Precio'"<10"{'Cantidad'"<10"{'Total'}\n")
        f.write("-" * 45 + "\n")

        # Datos con formateo
        total_general = 0
        for producto in productos:

            f.write(f"{producto['producto']"<15"producto['precio']"<10.2f"producto['cantidad']"<10"tal_producto"<10.2f"\n")

        f.write("-" * 45 + "\n")
        f.write(f"{'TOTAL GENERAL':"<35"tal_general"<10.2f"\n")

    print("✅ Tabla de productos creada")

    with open('productos_tabla.txt', 'r', encoding='utf-8') as f:
        tabla = f.read()
    print("\n" + tabla)

    # Limpiar
    for archivo in ['datos_basicos.txt', 'productos_tabla.txt']:
        if os.path.exists(archivo):
            os.remove(archivo)

    # Datos estructurados
    registros = [
        ('Ana', 25, 165.5, True),
        ('Carlos', 30, 175.2, False),
        ('María', 28, 160.8, True)
    ]

    with open('registros.bin', 'wb') as f:
        for nombre, edad, altura, activo in registros:
            # Empaquetar datos: string(10), int, float, char(bool)
            nombre_padded = nombre.ljust(10)[:10]  # Nombre de 10 caracteres máximo
            datos_empaquetados = struct.pack('10s i f ?',
                                          nombre_padded.encode('utf-8'),
                                          edad, altura, activo)
            f.write(datos_empaquetados)

    print(f"✅ Registros binarios escritos: {len(registros)} registros")

    # Leer registros binarios
    with open('registros.bin', 'rb') as f:
        print("\n📊 Registros leídos:")
        for i in range(len(registros)):
            datos = f.read(struct.calcsize('10s i f ?'))
            nombre, edad, altura, activo = struct.unpack('10s i f ?', datos)

            nombre = nombre.decode('utf-8').rstrip()

            print(f"  {nombre}: edad={edad}, altura={altura".1f"}, activo={activo}")

    for archivo in ['imagen.bin', 'registros.bin']:

def sistema_logging_avanzado():
    """Crea un sistema avanzado de logging con rotación y niveles."""

    print("\n=== SISTEMA DE LOGGING AVANZADO ===")

    class LoggerAvanzado:
        """Sistema avanzado de logging."""

        def __init__(self, archivo_log="app.log", nivel_minimo="DEBUG"):
            self.archivo_log = archivo_log
            self.nivel_minimo = nivel_minimo.upper()
            self.niveles = {
                'DEBUG': 0,
                'INFO': 1,
                'WARNING': 2,
                'ERROR': 3,
                'CRITICAL': 4
            }

        def _debe_loggear(self, nivel):
            """Determina si debe loggear según nivel mínimo."""
            return self.niveles.get(nivel.upper(), 0) >= self.niveles.get(self.nivel_minimo, 0)

        def _formatear_mensaje(self, nivel, mensaje, contexto=None):
            """Formatea el mensaje de log."""
            timestamp = datetime.now().strftime('%Y-%m-%d %H:%M:%S.%f')[:-3]

            # Obtener información del contexto si está disponible
            contexto_str = ""
            if contexto:
                contexto_str = f" | {contexto}"

            return f"[{timestamp}] [{nivel.upper()}] {mensaje}{contexto_str}\n"

        def debug(self, mensaje, contexto=None):
            """Loggea mensaje de debug."""
            if self._debe_loggear('DEBUG'):
                mensaje_formateado = self._formatear_mensaje('DEBUG', mensaje, contexto)
                self._escribir_log(mensaje_formateado)

        def info(self, mensaje, contexto=None):
            """Loggea mensaje informativo."""
            if self._debe_loggear('INFO'):
                mensaje_formateado = self._formatear_mensaje('INFO', mensaje, contexto)
                self._escribir_log(mensaje_formateado)

        def warning(self, mensaje, contexto=None):
            """Loggea advertencia."""
            if self._debe_loggear('WARNING'):
                mensaje_formateado = self._formatear_mensaje('WARNING', mensaje, contexto)
                self._escribir_log(mensaje_formateado)

        def error(self, mensaje, contexto=None):
            """Loggea error."""
            if self._debe_loggear('ERROR'):
                mensaje_formateado = self._formatear_mensaje('ERROR', mensaje, contexto)
                self._escribir_log(mensaje_formateado)

        def critical(self, mensaje, contexto=None):
            """Loggea error crítico."""
            if self._debe_loggear('CRITICAL'):
                mensaje_formateado = self._formatear_mensaje('CRITICAL', mensaje, contexto)
                self._escribir_log(mensaje_formateado)

        def _escribir_log(self, mensaje_formateado):
            """Escribe el mensaje formateado en el archivo."""
            try:
                with open(self.archivo_log, 'a', encoding='utf-8') as f:
                    f.write(mensaje_formateado)
            except Exception as e:
                print(f"❌ Error escribiendo en log: {e}")

        def limpiar_log(self, dias_antiguos=7):
            """Limpia entradas de log más antiguas que los días especificados."""
            try:
                if not os.path.exists(self.archivo_log):
                    return

                # Leer todas las líneas
                with open(self.archivo_log, 'r', encoding='utf-8') as f:
                    lineas = f.readlines()

                # Filtrar líneas recientes
                lineas_recientes = []
                corte_tiempo = datetime.now() - timedelta(days=dias_antiguos)

                for linea in lineas:
                    # Extraer timestamp de la línea
                    try:
                        # Formato: [2024-01-15 10:30:45.123] [NIVEL] mensaje
                        timestamp_str = linea[1:24]  # Extraer timestamp
                        timestamp = datetime.strptime(timestamp_str, '%Y-%m-%d %H:%M:%S.%f')

                        if timestamp > corte_tiempo:
                            lineas_recientes.append(linea)
                    except (ValueError, IndexError):
                        # Si no se puede parsear, mantener la línea
                        lineas_recientes.append(linea)

                # Reescribir archivo con líneas recientes
                with open(self.archivo_log, 'w', encoding='utf-8') as f:
                    f.writelines(lineas_recientes)

                print(f"✅ Log limpiado: mantenidas {len(lineas_recientes)}/{len(lineas)} líneas")

            except Exception as e:
                print(f"❌ Error limpiando log: {e}")

    # Demostración del logger avanzado
    logger = LoggerAvanzado("app_avanzado.log", nivel_minimo="INFO")

# Sistema avanzado de registro de actividades

import json
import threading
from datetime import datetime, timedelta
from collections import defaultdict
import os

class SistemaRegistroActividades:
    """Sistema completo para registrar actividades de aplicaciones."""

    def __init__(self, archivo_log="actividades.log", tamaño_maximo=10*1024*1024):  # 10MB
        self.archivo_log = archivo_log
        self.tamaño_maximo = tamaño_maximo
        self.lock = threading.Lock()  # Para acceso seguro desde múltiples hilos

        # Crear directorio si no existe
        os.makedirs(os.path.dirname(archivo_log), exist_ok=True)

    def registrar_actividad(self, usuario, accion, detalles=None, nivel="INFO"):
        """Registra una actividad en el sistema."""

        timestamp = datetime.now()

        entrada_log = {
            'timestamp': timestamp.isoformat(),
            'usuario': usuario,
            'accion': accion,
            'nivel': nivel.upper(),
            'detalles': detalles or {}
        }

        # Acceso seguro para múltiples hilos
        with self.lock:
            try:
                # Verificar tamaño del archivo y rotar si es necesario
                if os.path.exists(self.archivo_log):
                    tamaño_actual = os.path.getsize(self.archivo_log)
                    if tamaño_actual > self.tamaño_maximo:
                        self._rotar_archivo_log()

                # Escribir entrada
                with open(self.archivo_log, 'a', encoding='utf-8') as f:
                    f.write(json.dumps(entrada_log) + '\n')

            except Exception as e:
                print(f"❌ Error registrando actividad: {e}")

    def _rotar_archivo_log(self):
        """Rota el archivo de log cuando alcanza el tamaño máximo."""

        try:
            # Crear nombre para archivo antiguo
            timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
            archivo_antiguo = f"{self.archivo_log}.old.{timestamp}"

            # Renombrar archivo actual
            if os.path.exists(self.archivo_log):
                os.rename(self.archivo_log, archivo_antiguo)

            print(f"✅ Archivo de log rotado: {archivo_antiguo}")

        except Exception as e:
            print(f"❌ Error rotando archivo de log: {e}")

    def consultar_actividades(self, usuario=None, accion=None, fecha_inicio=None, fecha_fin=None):
        """Consulta actividades según criterios."""

        actividades = []

        try:
            with open(self.archivo_log, 'r', encoding='utf-8') as f:
                for linea in f:
                    try:
                        entrada = json.loads(linea.strip())

                        # Aplicar filtros
                        if usuario and entrada.get('usuario') != usuario:
                            continue

                        if accion and entrada.get('accion') != accion:
                            continue

                        if fecha_inicio or fecha_fin:
                            timestamp = datetime.fromisoformat(entrada['timestamp'])

                            if fecha_inicio and timestamp < fecha_inicio:
                                continue

                            if fecha_fin and timestamp > fecha_fin:
                                continue

                        actividades.append(entrada)

                    except json.JSONDecodeError:
                        continue  # Saltar líneas mal formateadas

        except FileNotFoundError:
            print(f"⚠️ Archivo de log '{self.archivo_log}' no encontrado")
        except Exception as e:
            print(f"❌ Error consultando actividades: {e}")

        return actividades

    def generar_reporte_actividades(self, fecha_inicio=None, fecha_fin=None):
        """Genera un reporte de actividades."""

        actividades = self.consultar_actividades(fecha_inicio=fecha_inicio, fecha_fin=fecha_fin)

        if not actividades:
            return "No hay actividades que mostrar"

        # Estadísticas
        actividades_por_usuario = defaultdict(int)
        actividades_por_accion = defaultdict(int)
        actividades_por_nivel = defaultdict(int)

        primera_fecha = None
        ultima_fecha = None

        for actividad in actividades:
            usuario = actividad.get('usuario', 'desconocido')
            accion = actividad.get('accion', 'desconocida')
            nivel = actividad.get('nivel', 'INFO')
            timestamp = datetime.fromisoformat(actividad['timestamp'])

            actividades_por_usuario[usuario] += 1
            actividades_por_accion[accion] += 1
            actividades_por_nivel[nivel] += 1

            if primera_fecha is None or timestamp < primera_fecha:
                primera_fecha = timestamp

            if ultima_fecha is None or timestamp > ultima_fecha:
                ultima_fecha = timestamp

        # Generar reporte
        reporte = []
        reporte.append("REPORTE DE ACTIVIDADES")
        reporte.append("=" * 50)
        reporte.append(f"Período: {primera_fecha} a {ultima_fecha}")
        reporte.append(f"Total actividades: {len(actividades)}")
        reporte.append("")

        reporte.append("POR USUARIO:")
        for usuario, cantidad in sorted(actividades_por_usuario.items()):
            reporte.append(f"  {usuario}: {cantidad}")

        reporte.append("\nPOR ACCIÓN:")
        for accion, cantidad in sorted(actividades_por_accion.items()):
            reporte.append(f"  {accion}: {cantidad}")

        reporte.append("\nPOR NIVEL:")
        for nivel, cantidad in sorted(actividades_por_nivel.items()):
            reporte.append(f"  {nivel}: {cantidad}")

        return "\n".join(reporte)

    def limpiar_actividades_antiguas(self, dias_antiguos=30):
        """Limpia actividades más antiguas que los días especificados."""

        fecha_corte = datetime.now() - timedelta(days=dias_antiguos)

        try:
            # Leer todas las actividades
            actividades_actuales = []
            actividades_antiguas = 0

            with open(self.archivo_log, 'r', encoding='utf-8') as f:
                for linea in f:
                    try:
                        entrada = json.loads(linea.strip())
                        timestamp = datetime.fromisoformat(entrada['timestamp'])

                        if timestamp > fecha_corte:
                            actividades_actuales.append(linea)
                        else:
                            actividades_antiguas += 1

                    except (json.JSONDecodeError, ValueError):
                        # Mantener líneas mal formateadas
                        actividades_actuales.append(linea)

            # Reescribir archivo
            with open(self.archivo_log, 'w', encoding='utf-8') as f:
                f.writelines(actividades_actuales)

            print(f"✅ Limpieza completada: eliminadas {actividades_antiguas} actividades antiguas")

        except Exception as e:
            print(f"❌ Error limpiando actividades: {e}")

            return self._generar_informe_csv(datos_ventas, nombre_base)
        elif formato == "json":
            return self._generar_informe_json(datos_ventas, nombre_base)
        elif formato == "html":
            return self._generar_informe_html(datos_ventas, nombre_base)
        else:
            raise ValueError(f"Formato no soportado: {formato}")

        """Genera informe en formato texto."""

        archivo = f"{self.directorio_salida}{nombre_base}.txt"

        with open(archivo, 'w', encoding='utf-8') as f:
            # Encabezado
            f.write("INFORME DE VENTAS\n")
            f.write("=" * 60 + "\n")
            f.write(f"Fecha del informe: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}\n")
            f.write(f"Productos incluidos: {len(datos)}\n\n")

{producto['producto']}{producto['ventas']}${producto['precio']".2f"}${ingresos".2f"}
TOTAL{total_ventas}-${total_ingresos".2f"}
\n\n")

            # Resumen estadístico
            resumen = self._calcular_resumen(datos)
            f.write("    
\n")
            f.write("        
Resumen Estadístico
\n")
            f.write(f"        
Producto más vendido: {resumen['producto_mas_vendido']}
\n")
            f.write(f"        
Promedio de ventas: {resumen['promedio_ventas']".1f"}
\n")
            f.write(f"        
Ingreso promedio: ${resumen['ingreso_promedio']".2f"}
\n")
            f.write("    
\n")

            f.write("
    
    
\n")
            f.write("\n")

        return archivo

    def _calcular_resumen(self, datos):
        """Calcula estadísticas resumen de los datos."""

        total_ventas = sum(p['ventas'] for p in datos)
        total_ingresos = sum(p['ventas'] * p['precio'] for p in datos)

        promedio_ventas = total_ventas / len(datos)
        ingreso_promedio = total_ingresos / len(datos)

        producto_mas_vendido = max(datos, key=lambda x: x['ventas'])['producto']

        return {
            'total_ventas': total_ventas,
            'total_ingresos': total_ingresos,
            'promedio_ventas': promedio_ventas,
            'ingreso_promedio': ingreso_promedio,
            'producto_mas_vendido': producto_mas_vendido
        }
